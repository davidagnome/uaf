// Declaration structure: scopes, nesting, prototypes, locals, globals, parameters.
//
// This file mostly pins the symbol table's ordering, which is invisible in the code segment but
// decides the layout of both the constants segment and the public-function index. Every list in
// the symbol table is PREPENDED to, so iteration is reverse-declaration order.

$VAR globalOne;
$VAR globalTwo;

// A zero-parameter prototype. Note the DOUBLE semicolon: compileFunctionDecl consumes one on the
// prototype path and CompileProgram then demands another. A prototype WITH parameters cannot be
// defined at all -- checkProtoParam is called on the wrong definition.
$FUNC forwardDeclared();;

$FUNC forwardDeclared() {
	$RETURN "resolved";
} forwardDeclared;

$FUNC twoParameters(first, second) {
	// The last formal declared gets frame offset 0, because actual parameters are pushed left
	// to right onto a downward-growing stack and the frame pointer lands on the last one.
	$RETURN first + "/" + second;
} twoParameters;

$FUNC withDefault(required, optional = "fallback") {
	// The default value is recorded on the formal parameter but the compiler still requires the
	// caller to supply every argument, so this reads as documentation rather than behaviour.
	$RETURN required + "/" + optional;
} withDefault;

$PUBLIC $FUNC callsOthers() {
	// Each user-function call pushes a default result BEFORE its arguments; that cell is the
	// slot the callee stores into. A statement-level call then pops the result.
	forwardDeclared();
	$RETURN twoParameters("a", "b") + "|" + withDefault("x", "y");
} callsOthers;

$PUBLIC $FUNC localVariables(param) {
	// Locals must precede all executable code: they share one BINOP_LOCALS cell whose operand is
	// incremented per declaration, and that cell has to be the function's second word.
	$VAR first;
	$VAR second;
	$VAR third;
	first = "1";
	second = first + "2";
	third = second + "3";
	$RETURN param + third;
} localVariables;

$PUBLIC $FUNC globalsAcrossCalls() {
	globalOne = "one";
	globalTwo = "two";
	$RETURN globalOne + globalTwo;
} globalsAcrossCalls;

$PUBLIC $FUNC nesting(outerParam) {
	// A nested function's body sits inside the enclosing function's code, so the enclosing body
	// gets a JUMP over it. Only the inner one that is declared $PUBLIC reaches the index, under
	// the name "nesting@innerPublic".
	$FUNC innerPrivate(x) {
		$RETURN "private:" + x;
	} innerPrivate;

	$PUBLIC $FUNC innerPublic(y) {
		$FUNC deeper(z) {
			$RETURN "deep:" + z;
		} deeper;
		$RETURN "public:" + y + "/" + deeper(y);
	} innerPublic;

	$RETURN innerPrivate(outerParam) + "|" + innerPublic(outerParam);
} nesting;

$PUBLIC $FUNC sharesInnerNames() {
	// The same inner name in a sibling scope must not collide -- that is the whole point of
	// local function definitions (src/GPDL/README.txt:114).
	$FUNC innerPrivate(x) {
		$RETURN "other:" + x;
	} innerPrivate;
	$RETURN innerPrivate("q");
} sharesInnerNames;
